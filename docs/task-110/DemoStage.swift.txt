import AppKit
import WebKit
import GRDB

// TEMPORARY (TASK-110): stages a demo world for website screenshots and
// executes commands appended to <dir>/cmd, one per line. Enabled only with
// DETOUR_DEMO_STAGE=<dir> and an isolated DETOUR_DATA_DIR. Not for commit.
@MainActor
final class DemoStage {
    static var current: DemoStage?

    let dir: String
    weak var wc: BrowserWindowController?
    private var offset: UInt64 = 0
    private var timer: Timer?
    private let store = TabStore.shared

    init(dir: String, wc: BrowserWindowController) {
        self.dir = dir
        self.wc = wc
        FileManager.default.createFile(atPath: dir + "/cmd", contents: Data())
        timer = Timer.scheduledTimer(withTimeInterval: 0.2, repeats: true) { [weak self] _ in
            MainActor.assumeIsolated { self?.poll() }
        }
        log("ready pid=\(ProcessInfo.processInfo.processIdentifier) window=\(wc.window?.windowNumber ?? -1)")
    }

    func log(_ s: String) {
        let line = "\(String(format: "%.2f", Date().timeIntervalSince1970)) \(s)\n"
        let path = dir + "/log.txt"
        if let h = FileHandle(forWritingAtPath: path) {
            h.seekToEndOfFile(); h.write(line.data(using: .utf8)!); h.closeFile()
        } else {
            FileManager.default.createFile(atPath: path, contents: line.data(using: .utf8))
        }
    }

    private func poll() {
        guard let h = FileHandle(forReadingAtPath: dir + "/cmd") else { return }
        defer { h.closeFile() }
        h.seek(toFileOffset: offset)
        let data = h.readDataToEndOfFile()
        guard !data.isEmpty, let text = String(data: data, encoding: .utf8), text.hasSuffix("\n") else { return }
        offset += UInt64(data.count)
        for line in text.split(separator: "\n") where !line.isEmpty {
            run(String(line))
        }
    }

    private func after(_ t: Double, _ b: @escaping () -> Void) {
        DispatchQueue.main.asyncAfter(deadline: .now() + t, execute: b)
    }

    private func space(named name: String) -> Space? {
        store.spaces.first { $0.name.lowercased() == name.lowercased() }
    }

    private func findTab(_ needle: String, in space: Space) -> BrowserTab? {
        let all = space.pinnedEntries.compactMap(\.tab) + (space.profile?.favorites.compactMap(\.tab) ?? []) + space.tabs
        return all.first { ($0.url?.absoluteString ?? "").contains(needle) || $0.title.contains(needle) }
    }

    // MARK: - Commands

    private func run(_ line: String) {
        let parts = line.split(separator: " ", maxSplits: 1).map(String.init)
        let cmd = parts[0], arg = parts.count > 1 ? parts[1] : ""
        guard let wc, let window = wc.window else { return }
        log("> \(line)")
        switch cmd {
        case "seed":
            seed()
        case "frame":
            let v = arg.split(separator: " ").compactMap { Double($0) }
            if v.count == 4 { window.setFrame(NSRect(x: v[0], y: v[1], width: v[2], height: v[3]), display: true) }
            log("frame \(window.frame) window=\(window.windowNumber)")
        case "to2x":
            // Move onto the Retina (virtual) display, at its top-left with a margin.
            guard let screen = NSScreen.screens.first(where: { $0.backingScaleFactor >= 2 }) else { log("no 2x screen"); return }
            let v = arg.split(separator: " ").compactMap { Double($0) }
            let size = v.count == 2 ? NSSize(width: v[0], height: v[1]) : window.frame.size
            let f = screen.frame
            window.setFrame(NSRect(x: f.minX + 40, y: f.maxY - 40 - size.height, width: size.width, height: size.height), display: true)
            log("frame \(window.frame) scale=\(window.backingScaleFactor) window=\(window.windowNumber)")
        case "visitall":
            // Host every tab once so background pages finish loading (TASK-112).
            guard let s = wc.activeSpace else { return }
            let tabs = s.pinnedEntries.compactMap(\.tab) + (s.profile?.favorites.compactMap(\.tab) ?? []) + s.tabs
            let back = wc.selectedTabID
            for (i, t) in tabs.enumerated() { after(Double(i) * 2.5) { wc.selectTab(id: t.id) } }
            after(Double(tabs.count) * 2.5 + 1) {
                if let back { wc.selectTab(id: back) }
                self.log("visitall done")
            }
        case "renamepinned":
            let kv = arg.split(separator: "|").map(String.init)
            guard kv.count == 2, let s = wc.activeSpace,
                  let e = s.pinnedEntries.first(where: { $0.pinnedURL.absoluteString.contains(kv[0]) }) else { log("no pinned \(arg)"); return }
            store.renamePinnedEntry(id: e.id, name: kv[1], in: s)
        case "deleteprofile":
            if let p = store.profiles.first(where: { $0.name == arg }) { _ = store.deleteProfile(id: p.id) } else { log("no profile \(arg)") }
        case "settingsrow":
            func tables(_ v: NSView) -> [NSTableView] { (v as? NSTableView).map { [$0] } ?? [] + v.subviews.flatMap(tables) }
            guard let root = SettingsWindowController.shared.window?.contentView, let row = Int(arg),
                  let table = tables(root).first else { log("no settings table"); return }
            table.selectRowIndexes(IndexSet(integer: row), byExtendingSelection: false)
        case "settingswidth":
            guard let sw = SettingsWindowController.shared.window, let w = Double(arg) else { return }
            var f = sw.frame; f.size.width = w
            sw.setFrame(f, display: true)
            log("settings window=\(sw.windowNumber) frame=\(sw.frame)")
        case "wallpaper":
            // wallpaper <RRGGBB> — solid desktop picture for captures. macOS applies it to
            // EVERY display and Space, not only `screen`, so only `demo.sh wallpaper`
            // (which snapshots the wallpaper store for `demo.sh wallpaper-restore`) may send it.
            guard FileManager.default.fileExists(atPath: dir + "/wallpaper-backup.plist") else {
                log("wallpaper refused: no wallpaper-backup.plist — use 'demo.sh wallpaper <hex>'"); return
            }
            guard let screen = NSScreen.screens.first(where: { $0.backingScaleFactor >= 2 }),
                  let v = Int(arg, radix: 16) else { log("wallpaper: no 2x screen / bad colour"); return }
            let color = NSColor(srgbRed: CGFloat((v >> 16) & 255) / 255, green: CGFloat((v >> 8) & 255) / 255,
                                blue: CGFloat(v & 255) / 255, alpha: 1)
            let image = NSImage(size: NSSize(width: 64, height: 64), flipped: false) { r in color.setFill(); r.fill(); return true }
            let url = URL(fileURLWithPath: dir + "/wallpaper-\(arg).png")
            if let tiff = image.tiffRepresentation, let rep = NSBitmapImageRep(data: tiff) {
                try? rep.representation(using: .png, properties: [:])?.write(to: url)
            }
            do { try NSWorkspace.shared.setDesktopImageURL(url, for: screen, options: [.imageScaling: NSImageScaling.scaleAxesIndependently.rawValue]) ; log("wallpaper set") }
            catch { log("wallpaper failed \(error)") }
        case "front":
            NSApp.activate(ignoringOtherApps: true)
            window.makeKeyAndOrderFront(nil)
        case "space":
            let words = arg.split(separator: " ").map(String.init)
            guard let s = space(named: words[0]) else { log("no space \(arg)"); return }
            if words.count > 1, words[1] == "animate" { wc.tabSidebar.animateToSpace(id: s.id) } else { wc.setActiveSpace(id: s.id) }
        case "select":
            guard let s = wc.activeSpace, let tab = findTab(arg, in: s) else { log("no tab \(arg)"); return }
            wc.selectTab(id: tab.id)
        case "folder":
            guard let s = wc.activeSpace, let f = s.pinnedFolders.first(where: { $0.name == arg }) else { log("no folder \(arg)"); return }
            store.togglePinnedFolderCollapsed(id: f.id, in: s)
        case "split":
            let urls = arg.split(separator: " ").map(String.init)
            guard let s = wc.activeSpace, urls.count == 2,
                  let a = findTab(urls[0], in: s), let b = findTab(urls[1], in: s) else { log("split: tabs not found"); return }
            store.createSplit(draggedTabID: b.id, targetTabID: a.id, edge: .right, in: s)
            wc.selectTab(id: a.id)
        case "appearance":
            NSApp.appearance = NSAppearance(named: arg == "dark" ? .darkAqua : .aqua)
        case "settings":
            // settings <spaces|profiles|general> — opens the pane on the 2x display.
            let sc = SettingsWindowController.shared
            if arg == "spaces" { sc.showSpacesPane() } else { sc.showWindow(nil) }
            if arg == "profiles", let tb = sc.window?.toolbar,
               let item = tb.items.first(where: { $0.itemIdentifier.rawValue.lowercased().contains("profile") }),
               let action = item.action {
                NSApp.sendAction(action, to: item.target ?? sc, from: item)
            }
            after(0.5) {
                guard let sw = sc.window else { return }
                sw.orderFront(nil)
                if let screen = NSScreen.screens.first(where: { $0.backingScaleFactor >= 2 }) {
                    let f = screen.frame
                    sw.setFrameTopLeftPoint(NSPoint(x: f.minX + 60, y: f.maxY - 60))
                }
                self.log("settings window=\(sw.windowNumber) frame=\(sw.frame)")
            }
        case "js":
            wc.selectedTab?.webView?.evaluateJavaScript(arg) { r, e in self.log("js -> \(String(describing: r)) \(e.map { "\($0)" } ?? "")") }
        case "key":
            // key <keyCode> <ctrl|shift|cmd|opt...>
            let w = arg.split(separator: " ").map(String.init)
            var flags: NSEvent.ModifierFlags = []
            for m in w.dropFirst() {
                switch m { case "ctrl": flags.insert(.control); case "shift": flags.insert(.shift)
                case "cmd": flags.insert(.command); case "opt": flags.insert(.option); default: break }
            }
            let chars = w[0] == "48" ? "\t" : ""
            if let code = UInt16(w[0]) {
                post(.flagsChanged, flags, code: 59)
                post(.keyDown, flags, code: code, chars: chars)
            }
        case "release":
            post(.flagsChanged, [], code: 59)
        case "info":
            for s in store.spaces {
                log("space \(s.name) \(s.emoji) profile=\(s.profile?.name ?? "-") pinned=\(s.pinnedEntries.count) tabs=\(s.tabs.count) favs=\(s.profile?.favorites.count ?? 0)")
            }
            log("frame \(window.frame) window=\(window.windowNumber) selected=\(wc.selectedTab?.title ?? "-")")
        default:
            extra(cmd, arg, wc: wc, window: window)
        }
    }

    private func post(_ type: NSEvent.EventType, _ flags: NSEvent.ModifierFlags, code: UInt16, chars: String = "") {
        guard let window = wc?.window,
              let e = NSEvent.keyEvent(with: type, location: .zero, modifierFlags: flags,
                                       timestamp: ProcessInfo.processInfo.systemUptime, windowNumber: window.windowNumber,
                                       context: nil, characters: chars, charactersIgnoringModifiers: chars,
                                       isARepeat: false, keyCode: code) else { return }
        NSApp.postEvent(e, atStart: false)
    }

    // MARK: - Seed

    private struct Plan {
        let name: String, emoji: String, color: String, profile: String
        let favorites: [String]
        let pinned: [String]
        let folders: [(String, [String])]
        let tabs: [String]
    }

    private func seed() {
        guard space(named: "Work") == nil else { log("already seeded"); return }
        let work = store.addProfile(name: "Work")
        let personal = store.addProfile(name: "Personal")
        let client = store.addProfile(name: "Client X")
        let profiles = ["Work": work, "Personal": personal, "Client X": client]

        let plans = [
            Plan(name: "Work", emoji: "🍊", color: "FF7600", profile: "Work",
                 favorites: ["https://slack.com", "https://linear.app", "https://www.notion.com", "https://github.com",
                             "https://www.figma.com", "https://www.linkedin.com", "https://www.youtube.com", "https://www.apple.com"],
                 pinned: ["https://developer.apple.com/documentation/webkit"],
                 folders: [("Design", ["https://developer.apple.com/design/human-interface-guidelines", "https://www.figma.com/community"]),
                           ("Research", ["https://webkit.org/blog/", "https://arstechnica.com"])],
                 tabs: ["https://github.com/tienshiao/detour", "https://en.wikipedia.org/wiki/WebKit",
                        "https://developer.apple.com/documentation/webkit/wkwebview",
                        "https://docs.github.com/en", "https://news.ycombinator.com"]),
            Plan(name: "Personal", emoji: "🌿", color: "7C5CFF", profile: "Personal",
                 favorites: ["https://www.youtube.com", "https://www.reddit.com", "https://en.wikipedia.org", "https://www.apple.com"],
                 pinned: [],
                 folders: [("News", ["https://theoldreader.com", "https://www.reddit.com", "https://news.ycombinator.com"])],
                 tabs: ["https://en.wikipedia.org/wiki/Tab_(interface)", "https://www.apple.com/mac/",
                        "https://www.youtube.com"]),
            Plan(name: "Reading", emoji: "📚", color: "1FB6A8", profile: "Personal",
                 favorites: [], pinned: [], folders: [],
                 tabs: ["https://en.wikipedia.org/wiki/Hypertext", "https://arstechnica.com",
                        "https://www.theverge.com", "https://en.wikipedia.org/wiki/Web_browser",
                        "https://daringfireball.net"]),
            Plan(name: "Side", emoji: "🎨", color: "E94F8C", profile: "Client X",
                 favorites: ["https://github.com", "https://linear.app"], pinned: [], folders: [],
                 tabs: ["https://github.com/tienshiao/detour/releases", "https://sparkle-project.org"]),
        ]

        let original = store.spaces
        // Pass 1: open every page so titles and favicons load before any of
        // them becomes a pinned entry or a favourite tile (both snapshot the
        // title at that moment).
        var pending: [() -> Void] = []
        var profileHasFavs = Set<UUID>()
        for plan in plans {
            let profile = profiles[plan.profile]!
            let space = store.addSpace(name: plan.name, emoji: plan.emoji, colorHex: plan.color, profileID: profile.id)
            if !profileHasFavs.contains(profile.id) {
                profileHasFavs.insert(profile.id)
                for url in plan.favorites {
                    let tab = store.addTab(in: space, url: URL(string: url)!)
                    pending.append {
                        self.store.detachTab(id: tab.id, from: space)
                        self.store.addFavorite(from: tab, profileID: profile.id)
                    }
                }
            }
            for url in plan.pinned {
                let tab = store.addTab(in: space, url: URL(string: url)!)
                pending.append { self.store.pinTab(id: tab.id, in: space) }
            }
            for (folderName, urls) in plan.folders {
                let tabs = urls.map { self.store.addTab(in: space, url: URL(string: $0)!) }
                pending.append {
                    let folder = self.store.addPinnedFolder(name: folderName, in: space)
                    for tab in tabs {
                        self.store.pinTab(id: tab.id, in: space)
                        if let entry = space.pinnedEntries.first(where: { $0.tab?.id == tab.id }) {
                            self.store.movePinnedTabToFolder(tabID: entry.id, folderID: folder.id, in: space)
                        }
                    }
                }
            }
            // Reversed: new tabs insert at the top, so the list reads in plan order.
            for url in plan.tabs.reversed() {
                _ = store.addTab(in: space, url: URL(string: url)!)
            }
        }
        for s in original { store.deleteSpace(id: s.id) }
        if let workSpace = space(named: "Work") { wc?.setActiveSpace(id: workSpace.id) }
        log("seeded tabs; waiting for pages to load")
        after(20) {
            pending.forEach { $0() }
            self.store.saveNow()
            self.log("seed done")
        }
    }
}
// MARK: - Capture session (agent-capture, TASK-292.6/.7/.8 media). Temporary, never committed.
extension DemoStage {
    static var quickFrame: NSRect?   // top-left (x, y from the 2x screen's top-left), w, h

    var retina: NSScreen? { NSScreen.screens.first { $0.backingScaleFactor >= 2 && $0.localizedName.contains("Capture") } ?? NSScreen.screens.first { $0.backingScaleFactor >= 2 } }

    /// Global Quartz (top-left origin) rect of an AppKit screen rect.
    func quartz(_ r: NSRect) -> String {
        let h = NSScreen.screens[0].frame.maxY
        return String(format: "%.0f %.0f %.0f %.0f", r.minX, h - r.maxY, r.width, r.height)
    }

    /// An AppKit frame from a rect given relative to the 2x screen's top-left.
    func onRetina(x: Double, y: Double, w: Double, h: Double) -> NSRect? {
        guard let f = retina?.frame else { return nil }
        return NSRect(x: f.minX + x, y: f.maxY - y - h, width: w, height: h)
    }

    func centered(_ size: NSSize, dy: Double = 0) -> NSRect? {
        guard let f = retina?.frame else { return nil }
        return NSRect(x: (f.midX - size.width / 2).rounded(), y: (f.midY - size.height / 2 + dy).rounded(), width: size.width, height: size.height)
    }

    func keyEvent(_ type: NSEvent.EventType, code: UInt16, chars: String, flags: NSEvent.ModifierFlags) {
        guard let w = NSApp.keyWindow ?? wc?.window,
              let e = NSEvent.keyEvent(with: type, location: .zero, modifierFlags: flags,
                                       timestamp: ProcessInfo.processInfo.systemUptime, windowNumber: w.windowNumber,
                                       context: nil, characters: chars, charactersIgnoringModifiers: chars,
                                       isARepeat: false, keyCode: code) else { log("no key window"); return }
        NSApp.postEvent(e, atStart: false)
    }

    static let keys: [String: (UInt16, String)] = [
        "return": (36, "\r"), "tab": (48, "\t"), "esc": (53, "\u{1b}"), "space": (49, " "), "delete": (51, "\u{7f}"),
        "t": (17, "t"), "n": (45, "n"), "w": (13, "w"), "y": (16, "y"), "l": (37, "l"), "o": (31, "o"), "down": (125, "\u{F701}"),
    ]

    func flags(_ words: [String]) -> NSEvent.ModifierFlags {
        var f: NSEvent.ModifierFlags = []
        for m in words {
            switch m { case "ctrl": f.insert(.control); case "shift": f.insert(.shift)
            case "cmd": f.insert(.command); case "opt": f.insert(.option); default: break }
        }
        return f
    }

    func allViews(_ v: NSView) -> [NSView] { [v] + v.subviews.flatMap(allViews) }

    func describe(_ v: NSView) -> String {
        var parts: [String] = []
        if let s = v.accessibilityLabel(), !s.isEmpty { parts.append("label=\(s)") }
        if let b = v as? NSButton { if !b.title.isEmpty { parts.append("title=\(b.title)") }
            if let d = b.image?.accessibilityDescription { parts.append("img=\(d)") } }
        if let t = v as? NSTextField, !t.stringValue.isEmpty { parts.append("text=\(t.stringValue)") }
        if let t = v.toolTip { parts.append("tip=\(t)") }
        if let iv = v as? NSImageView, let d = iv.image?.accessibilityDescription { parts.append("img=\(d)") }
        return parts.joined(separator: " ")
    }

    func extra(_ cmd: String, _ arg: String, wc: BrowserWindowController, window: NSWindow) {
        let words = arg.split(separator: " ").map(String.init)
        switch cmd {
        case "center":
            // center W H [dy]: the main window, mid 2x display.
            let v = words.compactMap { Double($0) }
            guard v.count >= 2, let r = centered(NSSize(width: v[0], height: v[1]), dy: v.count > 2 ? v[2] : 0) else { return }
            window.setFrame(r, display: true)
            log("frame quartz=\(quartz(window.frame)) window=\(window.windowNumber)")
        case "type":
            // type <text>: one character every 0.09 s to the key window.
            for (i, ch) in arg.enumerated() {
                after(Double(i) * 0.09) {
                    let c = String(ch)
                    let code: UInt16 = c == " " ? 49 : 0
                    self.keyEvent(.keyDown, code: code, chars: c, flags: [])
                    self.keyEvent(.keyUp, code: code, chars: c, flags: [])
                }
            }
        case "press":
            // press <name> [mods...]
            guard let k = Self.keys[words.first ?? ""] else { log("no key \(arg)"); return }
            let f = flags(Array(words.dropFirst()))
            keyEvent(.keyDown, code: k.0, chars: k.1, flags: f)
            keyEvent(.keyUp, code: k.0, chars: k.1, flags: f)
        case "ctrl":
            switch arg {
            case "down": keyEvent(.flagsChanged, code: 59, chars: "", flags: .control)
            case "tab": keyEvent(.keyDown, code: 48, chars: "\t", flags: .control); keyEvent(.keyUp, code: 48, chars: "\t", flags: .control)
            case "up": keyEvent(.flagsChanged, code: 59, chars: "", flags: [])
            default: break
            }
        case "winfo":
            for w in NSApp.windows where w.isVisible {
                log("win \(w.windowNumber) \(type(of: w)) ctl=\(w.windowController.map { "\(type(of: $0))" } ?? "-") key=\(w.isKeyWindow) quartz=\(quartz(w.frame)) title=\(w.title)")
            }
        case "frames":
            // frames <text>: views whose label/title/tooltip/text contains <text>, in global Quartz coordinates.
            for w in NSApp.windows where w.isVisible {
                guard let root = w.contentView?.superview ?? w.contentView else { continue }
                for v in allViews(root) where !v.isHiddenOrHasHiddenAncestor {
                    let d = describe(v)
                    guard !d.isEmpty, d.lowercased().contains(arg.lowercased()) else { continue }
                    let r = w.convertToScreen(v.convert(v.bounds, to: nil))
                    log("view \(type(of: v)) win=\(w.windowNumber) quartz=\(quartz(r)) \(d)")
                }
            }
        case "jsrect":
            // jsrect <css selector>: the element's rect in global Quartz coordinates (selected tab).
            guard let web = wc.selectedTab?.webView else { log("no webview"); return }
            let js = "(()=>{const e=document.querySelector(\(String(reflecting: arg)));if(!e)return 'null';const r=e.getBoundingClientRect();return JSON.stringify([r.x,r.y,r.width,r.height,e.textContent.slice(0,60)])})()"
            web.evaluateJavaScript(js) { r, e in
                guard let str = r as? String, let a = (try? JSONSerialization.jsonObject(with: Data(str.utf8))) as? [Any], a.count == 5,
                      let x = (a[0] as? NSNumber)?.doubleValue, let y = (a[1] as? NSNumber)?.doubleValue,
                      let w = (a[2] as? NSNumber)?.doubleValue, let h = (a[3] as? NSNumber)?.doubleValue else {
                    self.log("jsrect none \(String(describing: r)) \(String(describing: e))"); return }
                let local = NSRect(x: x, y: web.isFlipped ? y : web.bounds.height - y - h, width: w, height: h)
                let rr = web.window!.convertToScreen(web.convert(local, to: nil))
                self.log("jsrect quartz=\(self.quartz(rr)) center=\(Int(rr.midX)) \(Int(NSScreen.screens[0].frame.maxY - rr.midY)) text=\(a[4])")
            }
        case "load":
            if let u = URL(string: arg) { wc.selectedTab?.load(u, typed: false) }
        case "newtab":
            wc.newTab(nil)
        case "addtab":
            // addtab <url>: a tab in the active space, selected.
            guard let s = wc.activeSpace, let u = URL(string: arg) else { return }
            let t = store.addTab(in: s, url: u)
            wc.selectTab(id: t.id)
        case "closetab":
            guard let s = wc.activeSpace, let t = s.tabs.first(where: { ($0.url?.absoluteString ?? "").contains(arg) || $0.title.contains(arg) }) else { log("no tab \(arg)"); return }
            store.closeTab(id: t.id, in: s)
        case "qdframe":
            // qdframe X Y W H (relative to the 2x screen's top-left): where new Quick Detour windows go.
            let v = words.compactMap { Double($0) }
            guard v.count == 4 else { return }
            Self.quickFrame = onRetina(x: v[0], y: v[1], w: v[2], h: v[3])
            QuickDetourLauncher.shared.openWindow = { p, s, u, t in
                let c = QuickDetourWindowController.open(profile: p, sourceSpace: s, url: u, typed: t)
                if let f = DemoStage.quickFrame { c.window?.setFrame(f, display: false) }
                NSApp.activate()
                // As a hotkey-opened window over the app in front: in front of it (TASK-293.6 re-take).
                NSApp.activate(ignoringOtherApps: true)
                c.window?.makeKeyAndOrderFront(nil)
                c.window?.orderFrontRegardless()
            }
            log("qdframe \(String(describing: Self.quickFrame))")
        case "openext":
            // openext <bundleID> <url>: the external-link entry point, as a click in that app.

                if let f = Self.quickFrame { c.window?.setFrame(f, display: false) }

                NSApp.activate()
                c.window?.makeKeyAndOrderFront(nil)
            }
        case "quicklaunch":
            (NSApp.delegate as? AppDelegate)?.quickDetourHotKeyPressed()
            after(0.3) { if let p = QuickDetourLauncher.shared.panel { self.log("launcher quartz=\(self.quartz(p.frame))") } }
        case "routes":
            let lr = store.linkRoutes
            for r in lr.routes { lr.removeRoute(id: r.id) }
            var order = 0
            func add(_ m: LinkRouteMatch?, _ app: String?, _ t: LinkRouteTarget) {
                if lr.add(LinkRoute(match: m, sourceApp: app, target: t, sortOrder: order)) == nil { log("route refused") }
                order += 1
            }
            if words.contains("quarantine") { add(nil, "com.apple.TextEdit", .quarantine) }
            if words.contains("all") {
                if let work = space(named: "Work") { add(LinkRouteMatch(kind: .domainIs, pattern: "github.com"), nil, .space(work.id)) }

            }
            log("routes \(lr.routes.count)")
        case "private":
            // private <url> <W> <H>: a Private window with the page, mid 2x display.
            guard let u = URL(string: words.first ?? ""), let app = NSApp.delegate as? AppDelegate else { return }
            app.createNewWindowWithURL(u, incognito: true)
            let v = words.dropFirst().compactMap { Double($0) }
            if v.count == 2, let r = centered(NSSize(width: v[0], height: v[1])),
               let pw = NSApp.windows.first(where: { ($0.windowController as? BrowserWindowController)?.isIncognito == true }) {
                pw.setFrame(r, display: true)
                log("private window=\(pw.windowNumber) quartz=\(quartz(pw.frame))")
            }
        case "history":
            wc.showHistory(nil)
        case "archive":
            wc.showArchivedTabs(nil)
        case "archiveseed":
            // archiveseed: opens pages in the active space, then archives them on past days.
            guard let s = wc.activeSpace else { return }
            let plan: [(String, Double)] = [
                ("https://en.wikipedia.org/wiki/Hypertext", 2), ("https://www.apple.com/macbook-air/", 3),
                ("https://developer.mozilla.org/en-US/docs/Web/HTML", 5),
                ("https://en.wikipedia.org/wiki/Tim_Berners-Lee", 26), ("https://webkit.org/blog/", 28),
                ("https://www.youtube.com/results?search_query=swift+concurrency", 30),
                ("https://en.wikipedia.org/wiki/World_Wide_Web", 51), ("https://developer.apple.com/xcode/", 53),
            ]
            let tabs = plan.map { (store.addTab(in: s, url: URL(string: $0.0)!), $0.1) }
            log("archiveseed loading \(tabs.count)")
            after(15) {
                for (t, hours) in tabs { self.store.closeTab(id: t.id, in: s, archivedAt: Date(timeIntervalSinceNow: -hours * 3600)) }
                self.store.saveNow()
                self.log("archiveseed done")
            }
        case "agehistory":
            // agehistory <substring> <days>: moves that URL's visits back by <days>.
            guard words.count == 2, let days = Double(words[1]) else { return }
            let pattern = "%\(words[0])%"
            do {
                try HistoryDatabase.shared.dbQueue.write { db in
                    try db.execute(sql: "UPDATE historyVisit SET visitTime = visitTime - ? WHERE urlID IN (SELECT id FROM historyURL WHERE url LIKE ?)", arguments: [days * 86400, pattern])
                    try db.execute(sql: "UPDATE historyURL SET lastVisitTime = (SELECT MAX(visitTime) FROM historyVisit WHERE urlID = historyURL.id) WHERE url LIKE ?", arguments: [pattern])
                }
                log("aged \(words[0])")
            } catch { log("agehistory failed \(error)") }

            case "links": sc.showLinksPane()

            // sqlh|sqls <statement>: runs SQL on the history / session database (demo data only).
            do {
                let q = cmd == "sqlh" ? HistoryDatabase.shared.dbQueue : AppDatabase.shared.dbQueue
                try q.write { db in try db.execute(sql: arg); self.log("sql changed \(db.changesCount)") }
            } catch { log("sql failed \(error)") }
        case "hidemain":
            window.orderOut(nil)
        case "showmain":
            window.orderFront(nil)
        case "activate":
            NSApp.activate()
        case "closequick":
            for c in QuickDetourWindowController.openControllers { c.window?.performClose(nil) }
        case "quit":
            NSApp.terminate(nil)
        case "blocked":
            log("blocked=\(wc.displayTab?.blockedCount ?? -1) url=\(wc.displayTab?.url?.absoluteString ?? "-")")
        default:
            extra2(cmd, arg, words: words, wc: wc)
        }
    }

            guard words.count == 2, let p = store.profiles.first(where: { $0.name == words[0] }),
                  let u = URL(string: words[1]) else { log("installcrx: bad args"); return }
            let pid = p.id
            URLSession.shared.dataTask(with: u) { data, resp, err in
                let final = resp?.url ?? u
                DispatchQueue.main.async {
                    MainActor.assumeIsolated {
                        guard let data else { self.log("installcrx download failed \(String(describing: err))"); return }
                        do {
                            let verified = try CRXInstallGate.verify(crxData: data, downloadURL: final)
                            let c = try CRXInstallGate.admit(verified, crxData: data) { _ in nil }
                            defer { try? FileManager.default.removeItem(at: c.directory) }
                            var options = ExtensionInstaller.Options()
                            (options.source, options.updateURL) = ExtensionSource.classifyCRX(
                                fromWebStore: c.origin == .webStore, manifestUpdateURL: c.manifest.updateURL)
                            let ext = try ExtensionManager.shared.install(from: c.directory, publicKey: c.publicKey, options: options, turningOnIn: pid)
                            self.log("installed \(ext.id) \(ExtensionManager.shared.displayName(for: ext.id)) store=\(c.origin == .webStore) from=\(final.host ?? "-")")

extension DemoStage {
    /// Visible views in every window whose accessibility label or button title is exactly `text`.
    func views(exactly text: String) -> [NSView] {
        NSApp.windows.filter(\.isVisible).flatMap { w -> [NSView] in
            guard let root = w.contentView?.superview ?? w.contentView else { return [] }
            return allViews(root).filter { v in
                !v.isHiddenOrHasHiddenAncestor
                    && (v.accessibilityLabel() == text || (v as? NSButton)?.title == text)
            }
        }
    }

    func extra3(_ cmd: String, _ arg: String, words: [String], wc: BrowserWindowController) {
